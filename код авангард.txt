<script>
    const { Transformer } = window.markmap;
    const { Markmap } = window.markmapView;
    
    let mm = null;
    const transformer = new Transformer();
    
    function generateMap() {
        const input = document.getElementById('input').value.trim();
        
        if (!input) {
            alert('🌱 Введите текст для создания карты!');
            return;
        }
        
        const { root } = transformer.transform(input);
        const svg = document.getElementById('mindmap');
        svg.innerHTML = '';
        
        mm = Markmap.create(svg, {
            colorFreezeLevel: 2,
            duration: 500,
            maxWidth: 200,
            paddingX: 20
        }, root);
    }
    
    function loadExample() {
        document.getElementById('input').value = `# Иммануил Кант
## 📖 Биография
- 1724–1804 годы жизни
- Кёнигсберг (Калининград)
- Профессор университета
## 💭 Философия
- Критика чистого разума
- Категорический императив
- Трансцендентальный идеализм
## 🕰️ Эпоха
- Просвещение (XVIII век)
- Немецкая классическая философия
- Родоначальник нового мышления
## 🌍 Влияние
- Переворот в мировой мысли
- Основа современной этики
- Влияние на Гегеля, Фихте
## 🏛️ Наследие
- Труды по этике и эстетике
- Педагогические идеи
- Музей в Калининграде`;
        generateMap();
    }
    
    function clearInput() {
        document.getElementById('input').value = '';
document.getElementById('mindmap').innerHTML = '';
    }
    
    // Загрузить пример при старте
    window.onload = loadExample;
</script>
